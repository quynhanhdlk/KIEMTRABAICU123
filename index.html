// Nạp dữ liệu khởi tạo từ máy chủ Streamlit const serverData = window.INITIAL_CLOUD_DATA || {}; function getLocalData(key, fallback) { try { // Ưu tiên nạp từ dữ liệu máy chủ Streamlit truyền xuống if (key === 'saved_exams_list' && serverData.exams && serverData.exams.length > 0) { return serverData.exams; } if (key === 'student_submissions_list' && serverData.submissions && serverData.submissions.length > 0) { return serverData.submissions; } const l = localStorage.getItem(key); if (l !== null && l !== 'undefined') return JSON.parse(l); } catch(e){} return fallback; } function saveLocalData(key, value) { try { localStorage.setItem(key, JSON.stringify(value)); } catch(e){} // Tự động đồng bộ lên máy chủ Streamlit dùng chung if (key === 'saved_exams_list') { serverData.exams = value; if (value.length > 0) { syncToServer('save_exam', value[0]); } } else if (key === 'student_submissions_list') { serverData.submissions = value; if (value.length > 0) { syncToServer('save_submission', value[0]); } } } // Hàm ngầm gửi dữ liệu về máy chủ Streamlit function syncToServer(actionType, dataPayload) { try { const encodedPayload = encodeURIComponent(JSON.stringify(dataPayload)); fetch(`/?action=${actionType}&payload=${encodedPayload}`) .then(res => res.json()) .then(data => console.log("Đã đồng bộ lên máy chủ chung:", data)) .catch(err => console.warn("Lưu máy chủ lỗi:", err)); } catch(e) { console.warn(e); } }
