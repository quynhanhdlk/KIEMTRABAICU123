// ========================================================================= // AI CHẤM VÀ NHẬN XÉT: TỰ ĐỘNG THỬ MODEL & BÁO RÕ LỖI TỪ GOOGLE // ========================================================================= async function triggerAIGrading() { const sub = localStore.submissions.find(s => s.id === activeGradingId); if (!sub) return; const studentName = sub.studentName || 'Em'; const studentAnswer = (sub.answer || '').trim(); const rubricContent = (sub.examRubric || '').trim(); const examQuestion = sub.examTitle || 'Kiểm tra bài cũ'; if (!studentAnswer) { document.getElementById('grading-score-field').value = '0/10'; document.getElementById('grading-comment-field').value = `Chào ${studentName}, Cô chưa ghi nhận được câu trả lời của em trong bài kiểm tra này. Em hãy chú ý ôn tập kỹ lưỡng và chuẩn bị tốt hơn ở lần kiểm tra sau nhé!`; showToast("Học sinh chưa có câu trả lời!"); return; } let injectedKey = "{{GEMINI_API_KEY}}"; let apiKey = (injectedKey && !injectedKey.includes("{{") && injectedKey.length > 20) ? injectedKey.trim() : (localStorage.getItem('gemini_api_key_stored') || '').trim(); if (!apiKey) { apiKey = prompt("Chưa tìm thấy API Key hoặc Key bị khóa. Vui lòng dán mã API Key mới (AIzaSy...):"); if (apiKey) { apiKey = apiKey.trim(); localStorage.setItem('gemini_api_key_stored', apiKey); } else { showToast("Chưa có API Key!"); return; } } const btn = document.getElementById('btn-ai-grader'); btn.innerHTML = `🔄 Cô đang chấm bài theo barem...`; btn.disabled = true; const promptText = `Bạn là một giáo viên có tâm, giàu kinh nghiệm sư phạm, xưng "Cô" và gọi học sinh là "em". THÔNG TIN BÀI KIỂM TRA: - Đề bài: ${examQuestion} - BÀI LÀM CỦA HỌC SINH (${studentName}): "${studentAnswer}" *** BẢN QUY CHẾ BAREM & MỌI YÊU CẦU CHẤM CỦA GIÁO VIÊN ***: """ ${rubricContent || "Đánh giá mức độ hiểu bài và nắm vững kiến thức theo thang điểm 10."} """ NGUYÊN TẮC BẮT BUỘC: 1. Luôn chấm trên thang điểm 10 cho kiến thức đúng trọng tâm theo barem. 2. Học sinh nói qua micro nên hiểu bản chất ý đúng thay vì bắt bẻ từng từ ngữ. 3. Lời nhận xét xưng "Cô", gọi "em", dặn dò ân cần, khen ngợi cụ thể điểm mạnh và nhắc nhở điểm cần bổ sung. TRẢ VỀ DUY NHẤT một chuỗi JSON chuẩn (không dùng markdown \`\`\`json): {"score": "Điền điểm số (ví dụ 8.5/10)", "comment": "Lời nhận xét chi tiết của Cô"}`; const modelsToTry = ["gemini-2.5-flash", "gemini-1.5-flash"]; let lastErrorMsg = ""; let successResult = null; for (let modelName of modelsToTry) { try { const url = `https://generativelanguage.googleapis.com/v1beta/models/${modelName}:generateContent?key=${apiKey}`; const res = await fetch(url, { method: "POST", headers: { "Content-Type": "application/json" }, body: JSON.stringify({ contents: [{ parts: [{ text: promptText }] }], generationConfig: { temperature: 0.3, responseMimeType: "application/json" } }) }); if (res.ok) { const data = await res.json(); const rawOutput = data?.candidates?.[0]?.content?.parts?.[0]?.text || ''; let cleanJson = rawOutput.replace(/```json/gi, '').replace(/```/g, '').trim(); successResult = JSON.parse(cleanJson); break; } else { const errData = await res.json().catch(() => ({})); lastErrorMsg = `[Mã ${res.status}] ${errData?.error?.message || res.statusText}`; } } catch(e) { lastErrorMsg = e.message; } } btn.innerHTML = `🤖 AI Chấm & Nhận Xét Theo Đúng Barem Thang 10`; btn.disabled = false; if (successResult && successResult.score !== undefined && successResult.comment) { document.getElementById('grading-score-field').value = successResult.score; document.getElementById('grading-comment-field').value = successResult.comment; showToast("AI đã hoàn thành chấm bài!"); } else { // Nếu key sai hoặc bị Google vô hiệu hóa, xóa key cũ để cô nhập key mới localStorage.removeItem('gemini_api_key_stored'); alert("Google báo lỗi: " + lastErrorMsg + "\n\n(Nếu API Key bị quét và khóa trên GitHub, vui lòng tạo API Key mới dán vào Secrets của Streamlit nhé cô!)"); } }
